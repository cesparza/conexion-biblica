/* ═══════════ LÁMINAS: ESTUDIO INTERACTIVO CON IMAGEN (v185) ═══════════
   MECANISMO
   Una lámina (fuente/laminas.js) es una imagen partida en capas del mismo
   tamaño, una por símbolo, más los textos de cada parte. Se pinta como un SVG
   con una <image> por capa y un rectángulo invisible encima para tocarla (una
   imagen recibe el toque en toda su caja, también donde es transparente).
   Acercar es mover el viewBox del SVG; encender, un filtro; romper, partículas.
   Dos modos sobre los mismos datos: Estudiar (tocar, leer el texto y lo que
   significa, línea de tiempo, Daniel 7, Pruébate) y Presentar (pantalla oscura,
   un paso por toque o por flecha, para una clase).
   Las imágenes son archivos aparte (laminas/), no van dentro de index.html:
   se piden solo al abrir la lámina. En Jóvenes, traer.mjs cambia LAMINAS_DIR. */
const LAMINAS_DIR='/laminas/';
let lamId='', lamModo='', lamI=-1, lamPaso=0, lamReto=0, lamBien=0, lamListo=false;
const lamQuieto=()=>{try{return matchMedia('(prefers-reduced-motion: reduce)').matches;}catch(e){return false;}};
const LAM=()=>LAMINAS[lamId];
const lamParte=id=>LAM().partes.find(p=>p.id===id);
const lamSrc=f=>LAMINAS_DIR+LAM().dir+f+'.webp';
function lamTarjeta(cid){
  const L=LAMINAS[cid];
  if(!L)return '';
  return '<div class="lam-tarjeta"><div class="lam-mini"><img src="'+LAMINAS_DIR+L.dir+'babilonia.webp" alt="" loading="lazy">'+
    L.capas.map(k=>'<img src="'+LAMINAS_DIR+L.dir+k+'.webp" alt="" loading="lazy">').join('')+'</div>'+
    '<div class="lam-tx"><div class="lam-ant">Estudio interactivo</div><div class="lam-tit">'+esc(L.titulo)+'</div>'+
    '<p>Toca cada parte de la estatua: el texto, lo que significa, sus fechas y su fuente.</p>'+
    '<div class="lam-bots"><button type="button" class="btn ver" onclick="lamAbre(\''+cid+'\',\'estudiar\')">'+IC('lupa')+' Explorar</button>'+
    '<button type="button" class="btn tono" onclick="lamAbre(\''+cid+'\',\'presentar\')">'+IC('play')+' Presentar</button></div></div></div>';
}
function lamSVG(pref){
  const L=LAM(), C=L.cajas;
  const capa=id=>'<g class="lam-parte" data-p="'+id+'"><image href="'+lamSrc(id)+'" x="'+L.ox+'" y="0" width="768" height="1152"/>'+
    '<rect class="lam-hit" x="'+C[id][0]+'" y="'+C[id][1]+'" width="'+C[id][2]+'" height="'+C[id][3]+'" fill="transparent" onclick="lamToca(\''+id+'\')"/></g>';
  return '<svg class="lam-svg" id="'+pref+'svg" viewBox="0 0 '+L.w+' '+L.h+'" role="img" aria-label="'+esc(L.titulo)+'">'+
    '<defs><radialGradient id="'+pref+'sombra"><stop offset="0" stop-color="#000" stop-opacity=".55"/><stop offset="1" stop-color="#000" stop-opacity="0"/></radialGradient></defs>'+
    '<image class="lam-fondo" href="'+lamSrc('babilonia')+'" x="-473" y="0" width="2047" height="1152" preserveAspectRatio="xMidYMid slice"/>'+
    '<image class="lam-monte" href="'+lamSrc('monte')+'" x="-473" y="0" width="2047" height="1152" preserveAspectRatio="xMidYMid slice"/>'+
    '<ellipse cx="'+(L.ox+384)+'" cy="1138" rx="230" ry="26" fill="url(#'+pref+'sombra)"/>'+
    '<g class="lam-estatua">'+L.capas.map(capa).join('')+'</g><g class="lam-polvo"></g>'+
    '<g class="lam-parte lam-piedra" data-p="piedra" transform="translate(960 190)"><image href="'+lamSrc('piedra')+'" x="-110" y="-110" width="220" height="220"/>'+
    '<rect class="lam-hit" x="-110" y="-110" width="220" height="220" fill="transparent" onclick="lamToca(\'piedra\')"/></g></svg>';
}
/* Mueve el viewBox con suavidad. */
function lamAcerca(svg,dest,ms){
  if(!svg)return;
  const ini=svg.getAttribute('viewBox').split(/\s+/).map(Number);
  if(lamQuieto()||!ms){svg.setAttribute('viewBox',dest.join(' '));return;}
  const t0=performance.now();
  const paso=t=>{const k=Math.min(1,(t-t0)/ms),e=k<.5?2*k*k:1-Math.pow(-2*k+2,2)/2;
    svg.setAttribute('viewBox',ini.map((v,i)=>v+(dest[i]-v)*e).join(' '));if(k<1)requestAnimationFrame(paso);};
  requestAnimationFrame(paso);
}
function lamCaja(id,f){
  const L=LAM(),[x,y,w,h]=L.cajas[id],R=L.w/L.h;
  let W=Math.min(L.w,Math.max(w,h*R)*(f||1.35)),H=W/R;if(H>L.h){H=L.h;W=H*R;}
  return [Math.max(0,Math.min(L.w-W,x+w/2-W/2)),Math.max(0,Math.min(L.h-H,y+h/2-H/2)),W,H];
}
const lamTodo=()=>[0,0,LAM().w,LAM().h];
function lamLinea(hasta,on){
  const L=LAM(),A=-700,B=2080,pos=y=>(y-A)/(B-A)*100;let h='';
  L.tiempo.forEach((id,k)=>{const p=lamParte(id),w=Math.max(pos(p.a)-pos(p.de),2.4);
    const et={plata:'Plata',bronce:'Bronce',hierro:'Roma imperial',dedos:'Roma dividida',barro:'Roma papal · 1.260 años'}[id]||'';
    h+='<div class="lam-b'+(id==='barro'?' f2':'')+(k>=hasta?' oculta':'')+(on===id?' on':'')+'" style="--c:'+p.c+';left:'+pos(p.de)+'%;width:'+w+'%" title="'+esc(p.reino+', '+p.fechas)+'">'+(w>8?et:'')+'</div>';});
  if(hasta>=L.tiempo.length)h+='<div class="lam-b f2 tenue" style="--c:'+lamParte('barro').c+';left:'+pos(1798)+'%;width:'+(pos(2026)-pos(1798))+'%" title="Herida en 1798; sana y vuelve al final">herida · sana</div>';
  [[-605,'605 a.C.'],[-331,'331'],[-168,'168'],[476,'476'],[1798,'1798'],[2026,'hoy']].forEach(([y,t])=>h+='<div class="lam-m" style="left:'+pos(y)+'%">'+t+'</div>');
  return '<div class="lam-tl">'+h+'</div>';
}
function lamAbre(cid,modo){
  if(!LAMINAS[cid])return;
  lamId=cid;lamModo=modo;lamI=-1;lamPaso=0;lamReto=0;lamBien=0;lamListo=false;
  const el=document.getElementById('lam');
  el.hidden=false;el.className='lam'+(modo==='presentar'?' pres':'');
  document.body.classList.add('leyendo');
  if(modo==='presentar')lamPintaPres();else lamPintaEst();
  const f=el.querySelector('.lam-cerrar');if(f)f.focus();
}
function lamCierra(){
  const el=document.getElementById('lam');
  if(!el||el.hidden)return;
  el.hidden=true;el.innerHTML='';lamModo='';
  document.body.classList.remove('leyendo');
}
/* ── Estudiar ── */

  const L=LAM(),el=document.getElementById('lam');

function lamInicio(){
  const L=LAM(),svg=document.getElementById('lesvg');lamI=-1;
  if(svg){svg.classList.remove('atenua');svg.querySelectorAll('.lam-parte').forEach(x=>x.classList.remove('on'));
    svg.querySelector('.lam-monte').style.opacity='0';svg.querySelector('.lam-estatua').style.opacity='1';lamAcerca(svg,lamTodo(),650);}
  const v=document.getElementById('lamVuelve');if(v)v.hidden=true;
  document.getElementById('lamPanel').innerHTML='<div class="lam-p1">Escoge una parte</div><div class="lam-chips">'+
    L.partes.map((p,i)=>'<button type="button" class="lam-chip" onclick="lamElige('+i+')"><span class="lam-sw" style="--c:'+p.c+'"></span>'+esc(p.nombre)+'</button>').join('')+'</div>'+
    '<div class="lam-capa t"><div class="lam-rot">'+L.inicio.ref+'</div><p><q>'+esc(L.inicio.texto)+'</q></p></div><p class="lam-nota">'+L.inicio.nota+'</p>';
  document.getElementById('lamTl').innerHTML=lamLinea(99,null);
}
function lamToca(id){
  if(lamModo==='presentar'){lamPres(lamPaso+1);return;}
  lamElige(LAM().partes.findIndex(p=>p.id===id));
}
function lamElige(i){
  const L=LAM(),p=L.partes[i],svg=document.getElementById('lesvg');
  if(!p||!svg)return;
  lamI=i;
  svg.classList.add('atenua');
  svg.querySelectorAll('.lam-parte').forEach(x=>x.classList.toggle('on',x.getAttribute('data-p')===p.id));
  svg.querySelector('.lam-monte').style.opacity=p.id==='monte'?'1':'0';
  svg.querySelector('.lam-estatua').style.opacity=p.id==='monte'?'0':'1';
  lamAcerca(svg,p.id==='monte'?lamTodo():lamCaja(p.id),650);
  document.getElementById('lamVuelve').hidden=false;
  document.getElementById('lamPanel').innerHTML=
    '<div class="lam-pcab"><div class="lam-metal" style="--c:'+p.c+'"></div><div><div class="lam-p1">'+(i+1)+' de '+L.partes.length+' · '+esc(p.nombre)+'</div>'+
    '<div class="lam-p2">'+esc(p.reino)+'</div><div class="lam-fechas">'+esc(p.fechas)+'</div></div></div>'+
    '<div class="lam-capa t"><div class="lam-rot">Lo que dice el texto · RV1995</div><p><q>'+esc(p.texto)+'</q> <span class="lam-ref">'+p.ref+'</span></p><p><q>'+esc(p.texto2)+'</q> <span class="lam-ref">'+p.ref2+'</span></p></div>'+
    '<div class="lam-capa i"><div class="lam-rot">Lo que significa · interpretación</div>'+p.interp.map(x=>'<p>'+x+'</p>').join('')+'<div class="lam-fu">Fuente: '+p.fuente+'</div></div>'+
    '<div class="lam-nav"><button type="button" class="btn tono" '+(i>0?'':'disabled ')+'onclick="lamElige('+(i-1)+')">‹ Anterior</button>'+
    '<button type="button" class="btn ver" '+(i<L.partes.length-1?'':'disabled ')+'onclick="lamElige('+(i+1)+')">Siguiente ›</button></div>';
  document.getElementById('lamTl').innerHTML=lamLinea(99,p.id);
}
function lamPintaReto(){
  const L=LAM(),el=document.getElementById('lamReto');if(!el)return;
  if(lamReto>=L.reto.length){el.innerHTML='<div class="lam-q">Terminaste: '+lamBien+' de '+L.reto.length+'.</div><button type="button" class="btn tono" onclick="lamOtraVez()">Otra vez</button>';return;}
  const x=L.reto[lamReto];lamListo=false;
  el.innerHTML='<div class="lam-nota">Pregunta '+(lamReto+1)+' de '+L.reto.length+'</div><div class="lam-q">'+esc(x.q)+'</div><div class="lam-ops">'+
    L.partes.filter(p=>p.id!=='monte').map(p=>'<button type="button" data-id="'+p.id+'" onclick="lamResponde(\''+p.id+'\')"><span class="lam-sw" style="--c:'+p.c+'"></span>'+esc(p.nombre)+'</button>').join('')+
    '</div><div class="lam-fb" id="lamFb" aria-live="polite"></div>';
}
function lamOtraVez(){lamReto=0;lamBien=0;lamPintaReto();}
function lamResponde(id){
  if(lamListo)return;lamListo=true;
  const x=LAM().reto[lamReto],ok=id===x.ok,el=document.getElementById('lamReto');if(ok)lamBien++;
  el.querySelectorAll('.lam-ops button').forEach(b=>{const k=b.getAttribute('data-id');if(k===x.ok)b.classList.add('ok');else if(k===id)b.classList.add('mal');});
  document.getElementById('lamFb').innerHTML=(ok?'<strong>Bien.</strong> ':'<strong>Era otra.</strong> ')+esc(x.e);
  setTimeout(()=>{lamReto++;lamPintaReto();},2200);
}
/* ── Presentar ── */
const lamPasos=()=>[{tipo:'portada'},...LAM().tiempo.map(id=>({tipo:'parte',id})),{tipo:'parte',id:'piedra',anim:'cae'},{tipo:'parte',id:'monte',anim:'monte'},{tipo:'cierre'}];
function lamPintaPres(){
  const el=document.getElementById('lam');
  el.innerHTML='<div class="lam-pesc">'+lamSVG('lp')+'</div><div class="lam-ptx" id="lamPtx"></div>'+
    '<div class="lam-pie"><div class="lam-desliza" id="lamPtl"></div><div class="lam-ctl">'+
    '<button type="button" id="lamAnt" aria-label="Anterior" onclick="lamPres(lamPaso-1)">‹</button><span id="lamCuenta"></span>'+
    '<button type="button" id="lamSig" aria-label="Siguiente" onclick="lamPres(lamPaso+1)">›</button>'+
    '<button type="button" class="lam-cerrar" onclick="lamCierra()">Salir</button></div></div>';
  lamPres(0);
}
function lamPres(k){
  const pasos=lamPasos();if(k<0||k>=pasos.length)return;
  const dir=k>lamPaso?1:-1;lamPaso=k;const s=pasos[k],L=LAM(),svg=document.getElementById('lpsvg');
  const vistos=pasos.slice(1,k+1).filter(x=>x.tipo==='parte').map(x=>x.id),rotos=vistos.includes('monte');
  svg.querySelectorAll('.lam-estatua .lam-parte').forEach(x=>{const id=x.getAttribute('data-p');
    x.style.opacity=rotos?'0':(s.tipo==='portada'?'.25':vistos.includes(id)?'1':'.08');x.classList.toggle('on',s.id===id);});
  const piedra=svg.querySelector('.lam-piedra');
  piedra.style.opacity=s.id==='piedra'?'1':'0';
  svg.querySelector('.lam-monte').style.opacity=(rotos||s.tipo==='cierre')?'1':'0';
  piedra.setAttribute('transform','translate(960 190)');
  if(s.anim==='cae'&&dir>0)lamCae(piedra);
  else if(s.id==='piedra')piedra.setAttribute('transform','translate('+L.golpe[0]+' '+L.golpe[1]+') rotate(40)');
  if(s.anim==='monte'&&dir>0)lamPolvo(svg);
  if(s.tipo==='parte'&&L.tiempo.includes(s.id))lamAcerca(svg,lamCaja(s.id,1.9),900);else lamAcerca(svg,lamTodo(),900);
  document.getElementById('lamPtl').innerHTML=lamLinea(s.tipo==='portada'?0:vistos.filter(id=>L.tiempo.includes(id)).length,s.id||null);
  const T=document.getElementById('lamPtx');
  if(s.tipo==='portada')T.innerHTML='<div class="k">Daniel 2</div><div class="h">El sueño de Nabucodonosor</div><div class="v">«'+esc(L.inicio.texto)+'»</div><div class="r">'+L.inicio.ref+' · RV1995</div><div class="s">Cuatro reinos, Roma en tres fases y un reino que no será jamás destruido.</div>';
  else if(s.tipo==='cierre')T.innerHTML='<div class="k">Daniel 2:44</div><div class="h">'+esc(L.cierre.titulo)+'</div><div class="v">«'+esc(L.cierre.texto)+'»</div><div class="r">'+L.cierre.ref+' · RV1995</div><div class="s">'+esc(L.cierre.nota)+'</div>';
  else{const p=lamParte(s.id);T.innerHTML='<div class="k">'+esc(p.nombre)+'</div><div class="h">'+esc(p.reino)+'</div><div class="f">'+esc(p.fechas)+'</div><div class="v">«'+esc(p.texto2)+'»</div><div class="r">Daniel '+p.ref2+' · RV1995</div><div class="s">'+esc(p.corto)+'</div><div class="fu">Fuente: '+p.fuente+'</div>';}
  document.getElementById('lamCuenta').textContent=(k+1)+' / '+pasos.length;
  document.getElementById('lamAnt').disabled=k===0;document.getElementById('lamSig').disabled=k===pasos.length-1;
}
function lamCae(piedra){
  const [gx,gy]=LAM().golpe;
  if(lamQuieto()){piedra.setAttribute('transform','translate('+gx+' '+gy+') rotate(40)');return;}
  const t0=performance.now(),ms=1100;
  const f=t=>{const k=Math.min(1,(t-t0)/ms),e=k*k;
    piedra.setAttribute('transform','translate('+(960+(gx-960)*e)+' '+(190+(gy-190)*e)+') rotate('+(k*300)+')');
    if(k<1)requestAnimationFrame(f);else{const est=piedra.parentNode.querySelector('.lam-estatua');
      if(est&&est.animate)est.animate([{transform:'translate(0,0)'},{transform:'translate(-12px,4px)'},{transform:'translate(12px,-2px)'},{transform:'translate(0,0)'}],{duration:380});}};
  requestAnimationFrame(f);
}
/* La estatua se hace «tamo» y se lo lleva el viento (2:35). */
function lamPolvo(svg){
  const capa=svg.querySelector('.lam-polvo');
  if(lamQuieto()){capa.innerHTML='';return;}
  const L=LAM(),col={oro:'#E9C255',plata:'#C9D1DA',bronce:'#C27A42',hierro:'#56606C',barro:'#B9805E',dedos:'#B9805E'};let h='';
  L.capas.forEach(id=>{const [x,y,w,hh]=L.cajas[id],n=id==='dedos'?20:70;
    for(let i=0;i<n;i++){const px=x+Math.random()*w,py=y+Math.random()*hh,r=5+Math.random()*13;
      h+='<path d="M'+px+' '+(py-r)+'l'+r+' '+(r*1.6)+'h'+(-r*2)+'z" fill="'+col[id]+'"/>';}});
  capa.innerHTML=h;
  capa.querySelectorAll('path').forEach(el=>{if(el.animate)el.animate([{transform:'translate(0,0) rotate(0)',opacity:1},{transform:'translate('+(450+Math.random()*900)+'px,'+(-160+Math.random()*220)+'px) rotate('+(Math.random()*540)+'deg)',opacity:0}],
    {duration:1400+Math.random()*900,delay:Math.random()*350,easing:'cubic-bezier(.3,.1,.3,1)',fill:'forwards'});});
  setTimeout(()=>{capa.innerHTML='';},2800);
}
document.addEventListener('keydown',e=>{
  if(!lamModo)return;
  if(e.key==='Escape'){lamCierra();return;}
  if(lamModo!=='presentar')return;
  if(['ArrowRight','ArrowDown','PageDown',' ','Enter'].includes(e.key)){e.preventDefault();lamPres(lamPaso+1);}
  else if(['ArrowLeft','ArrowUp','PageUp','Backspace'].includes(e.key)){e.preventDefault();lamPres(lamPaso-1);}
});
